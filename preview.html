你好，帮我写一段 React Native 的 FlatList 示例

好的，这是一个简单的 FlatList 示例：

import React from 'react';
import { FlatList, View, Text } from 'react-native';

const items = [
  { id: '1', title: 'Item 1' },
  { id: '2', title: 'Item 2' },
  { id: '3', title: 'Item 3' },
];

export function MyList() {
  return (
    <FlatList
      data={items}
      keyExtractor={(item) => item.id}
      renderItem={({ item }) => (
        <View>
          <Text>{item.title}</Text>
        </View>
      )}
    />
  );
}
